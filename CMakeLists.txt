cmake_minimum_required(VERSION 3.21)

# PersonalCEF intentionally supports only Debug and Release.
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
set(CMAKE_SUPPRESS_REGENERATION ON)

project(PersonalCEF LANGUAGES C CXX)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

if(NOT WIN32)
  message(FATAL_ERROR "PersonalCEF r9 is a Windows / Visual Studio 2022 project.")
endif()

# CEF_ROOT may arrive from cmd.exe using native Windows backslashes
# (for example D:\\soft\\PersonalCEF\\deps\\cef\\...).  CMake treats
# backslashes inside later string expansions as escape characters (\s, \P,
# etc.), so normalize the SDK path once at the project boundary before any
# CEF macro sees it.  Keep the official CEF SDK itself completely untouched.
if(DEFINED CEF_ROOT)
  file(TO_CMAKE_PATH "${CEF_ROOT}" PERSONALCEF_CEF_ROOT_NORMALIZED)
  set(CEF_ROOT "${PERSONALCEF_CEF_ROOT_NORMALIZED}" CACHE PATH
      "CEF binary distribution root" FORCE)
  unset(PERSONALCEF_CEF_ROOT_NORMALIZED)
endif()

if(NOT DEFINED CEF_ROOT OR NOT EXISTS "${CEF_ROOT}/cmake/FindCEF.cmake")
  message(FATAL_ERROR "CEF_ROOT is missing or invalid. Run START-HERE.cmd first.")
endif()

message(STATUS "PersonalCEF normalized CEF_ROOT: ${CEF_ROOT}")

# We use the standard non-installer CEF binary distribution and keep sandbox
# disabled, matching the previous PersonalCEF builds.
set(USE_SANDBOX OFF CACHE BOOL "CEF sandbox" FORCE)
set(USE_INSTALLER OFF CACHE BOOL "CEF installer mode" FORCE)

set(CMAKE_MODULE_PATH "${CEF_ROOT}/cmake" ${CMAKE_MODULE_PATH})
find_package(CEF REQUIRED)

# Fixed application output layout. CMake/OBJ files stay under build/vs2022.
set(PERSONALCEF_OUTPUT_DIR "${CMAKE_SOURCE_DIR}/bin" CACHE PATH "PersonalCEF runtime output root" FORCE)

# Official CEF wrapper only. No ceftests/cefsimple/gmock/gtest projects.
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} "${CMAKE_BINARY_DIR}/libcef_dll_wrapper")

# This folder is generated by tools/sync_cefclient.ps1 from the exact CEF
# distribution version. It contains only official cefclient + shared source.
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/tests/cefclient/CMakeLists.txt")
  message(FATAL_ERROR "Generated cefclient source is missing. Run START-HERE.cmd.")
endif()
add_subdirectory(tests/cefclient "${CMAKE_BINARY_DIR}/app")

# F5 always starts the real application target.
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT PersonalCEF)

PRINT_CEF_CONFIG()

# r10.2: r6-compatible temporary login shell. Official cefclient creates libcef_lib before this subdirectory is added.
add_subdirectory(login_helper)
